```python
from manim import *
import numpy as np
from audio_mixin import AudioMixin


class VectorIntuitionScene(AudioMixin, Scene):
    def construct(self):
        title = Text("Understanding Vectors", font_size=32).to_edge(UP, buff=0.5)
        self.play_with_audio(Write(title), "Vectors represent direction and magnitude")
        
        grid = NumberPlane().set_opacity(0.4)
        self.play(Create(grid))
        
        vector = Arrow(ORIGIN, [3, 2, 0], buff=0).set_color(YELLOW)
        self.play_with_audio(GrowArrow(vector), "This arrow shows a vector pointing from origin")
        
        label = MathTex(r"\vec{v} = \begin{bmatrix} 3 \\ 2 \end{bmatrix}").next_to(vector, RIGHT)
        self.play(Write(label))
        self.wait(1)
        
        self.play(FadeOut(*self.mobjects))
        self.wait(0.5)


class VectorOperationsScene(AudioMixin, Scene):
    def construct(self):
        title = Text("Vector Addition", font_size=32).to_edge(UP, buff=0.5)
        self.play(Write(title))
        
        v1 = Arrow(ORIGIN, [2, 1, 0], buff=0).set_color(BLUE)
        v2 = Arrow([2, 1, 0], [4, 3, 0], buff=0).set_color(GREEN)
        v_sum = Arrow(ORIGIN, [4, 3, 0], buff=0).set_color(RED)
        
        self.play(GrowArrow(v1))
        self.play_with_audio(GrowArrow(v2), "Adding a second vector")
        self.play_with_audio(GrowArrow(v_sum), "The sum is the diagonal")
        
        self.wait(2)
        self.play(FadeOut(*self.mobjects))
        self.wait(0.5)
```
